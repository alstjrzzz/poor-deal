56 @GetMapping("/")
57 public String getPosts(
58 @RequestParam(defaultValue = "1") int page,
59 @ModelAttribute("searchCondition") PostSearchCondition condition,
60 Model model, Principal principal) {
61
62 int totalCount = postService.getPostCount(condition);
63 int totalPages = (int) Math.ceil((double) totalCount / PAGESIZE);
64 if (totalPages == 0) totalPages = 1;
65
66 int offset = (page - 1) * PAGESIZE;
67 int startPage = ((page - 1) / BLOCK_SIZE) * BLOCK_SIZE + 1;
68 int endPage = startPage + BLOCK_SIZE - 1;
69 if (endPage > totalPages) {
70 endPage = totalPages;
71 }
72
73 List<PostSummary> postList = postService.getPagedPostsSummary(offset, PAGESIZE, condition);
74
75 model.addAttribute("postList", postList);
76 model.addAttribute("currentPage", page);
77 model.addAttribute("totalPages", totalPages);
78 model.addAttribute("startPage", startPage);
79 model.addAttribute("endPage", endPage);
80
81 if (principal != null) {
82 Long point = memberService.findByUserId(principal.getName()).getPoint();
83 model.addAttribute("myPoint", point);
84 }

86 return "post/list";

133 @GetMapping("/post/{postId}")
134 public String getPost(@PathVariable("postId") Long postId,
135 @ModelAttribute("searchCondition") PostSearchCondition condition,
136 @RequestParam(defaultValue = "1") int page,
137 Model model, Principal principal) {
138
139 PostResponse post = postService.getPost(postId);
140 List<CommentAndRepliesResponse> commentsAndReplies = postService.getCommentsAndReplies(postId);
141 int totalCommentCount = commentsAndReplies.stream()
142 .mapToInt(cr -> 1 + (cr.getReplies() != null ? cr.getReplies().size() : 0))
143 .sum();
144
145 boolean isApplied = false;
146 Long loginId = memberService.findByUserId(principal.getName()).getId();
147 if (!loginId.equals(post.getAuthorId())) {
148 isApplied = mailService.hasApplied(loginId, postId);
149 }
150
151 model.addAttribute("post", post);
152 model.addAttribute("commentsAndReplies", commentsAndReplies);
153 model.addAttribute("totalCommentCount", totalCommentCount);
154 model.addAttribute("isApplied", isApplied);
155 model.addAttribute("loginId", loginId);
156 model.addAttribute("currentPage", page);
157
158 return "post/detail";
159 }
